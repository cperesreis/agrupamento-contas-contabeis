SELECT IDCLIFOR,
       NOME,
       IDRECEBIMENTO,
       DESCRRECEBIMENTO,
       IDTITULO,
       DIGITOTITULO,
       SERIENOTA,
       DTPAGAMENTO,
       VALTITULO,
       VALPAGAMENTOTITULO,
       VALJUROSCOBRADO,
       VALJUROSPOSTERGADO,
       VALDESCCONCEDIDOTITULO,
       DTVENCIMENTO,
       IDCTACREDITO,
       DESCRCTACREDITO,
       IDCTADEBITO,
       CBEB.DESCRCTACONTABIL AS DESCRDEB
FROM (
         SELECT CLIENTE_FORNECEDOR.IDCLIFOR,
                CLIENTE_FORNECEDOR.NOME,
                FORMA_PAGREC.IDRECEBIMENTO,
                FORMA_PAGREC.DESCRRECEBIMENTO,
                CONTAS_PAGAR_BAIXAS.IDTITULO,
                CONTAS_PAGAR_BAIXAS.DIGITOTITULO,
                CONTAS_PAGAR_BAIXAS.SERIENOTA,
                CONTAS_PAGAR_BAIXAS.DTPAGAMENTO,
                CONTAS_PAGAR.VALTITULO,
                CONTAS_PAGAR_BAIXAS.VALPAGAMENTOTITULO,
                CONTAS_PAGAR_BAIXAS.VALJUROSCOBRADO,
                CONTAS_PAGAR_BAIXAS.VALJUROSPOSTERGADO,
                CONTAS_PAGAR_BAIXAS.VALDESCCONCEDIDOTITULO,
                CONTAS_PAGAR.DTVENCIMENTO,
                CONTAS_PAGAR.IDCTACONTABIL                                AS IDCTACREDITO,
                CTA.DESCRCTACONTABIL                                      AS DESCRCTACREDITO,
                COALESCE(IDCTACONTABILCONTRAPARTIDA,
                         UF_FRST_CPAG_PESQ(CONTAS_PAGAR.IDEMPRESA, CONTAS_PAGAR.IDPLANILHA, 'D',
                                           CONTAS_PAGAR.ORIGEMMOVIMENTO)) AS IDCTADEBITO
         FROM CLIENTE_FORNECEDOR,
              CONTAS_PAGAR_BAIXAS,
              CONTAS_PAGAR
                  JOIN DBA.CONTABIL_PLANO_CONTAS AS CTA ON CONTAS_PAGAR.IDCTACONTABIL = CTA.IDCTACONTABIL,
              FORMA_PAGREC
         WHERE CLIENTE_FORNECEDOR.IDCLIFOR = CONTAS_PAGAR_BAIXAS.IDCLIFOR
           AND FORMA_PAGREC.IDRECEBIMENTO = CONTAS_PAGAR.IDPAGAMENTO
           AND CONTAS_PAGAR_BAIXAS.IDEMPRESA = CONTAS_PAGAR.IDEMPRESA
           AND CONTAS_PAGAR_BAIXAS.IDCLIFOR = CONTAS_PAGAR.IDCLIFOR
           AND CONTAS_PAGAR_BAIXAS.IDTITULO = CONTAS_PAGAR.IDTITULO
           AND CONTAS_PAGAR_BAIXAS.DIGITOTITULO = CONTAS_PAGAR.DIGITOTITULO
           AND CONTAS_PAGAR_BAIXAS.SERIENOTA = CONTAS_PAGAR.SERIENOTA
           AND CONTAS_PAGAR_BAIXAS.IDEMPRESA IN (:RA_IDEMPRESA)
           AND FORMA_PAGREC.IDRECEBIMENTO IN (:RA_IDRECEBIMENTO)
           AND CONTAS_PAGAR_BAIXAS.IDCLIFOR = :RA_IDCLIFOR
           AND CONTAS_PAGAR_BAIXAS.DTPAGAMENTO BETWEEN :RA_DTINI AND :RA_DTFIM
           AND NOT EXISTS
             (
                 SELECT 0
                 FROM CONTAS_PAGAR_BAIXAS AS CPB
                          JOIN CONTAS_PAGAR AS CP ON
                     (CPB.IDEMPRESABAIXA = CP.IDEMPRESA AND
                      CPB.IDPLANILHA = CP.IDPLANILHA)
                 WHERE CONTAS_PAGAR_BAIXAS.IDEMPRESA = CPB.IDEMPRESA
                   AND CONTAS_PAGAR_BAIXAS.IDCLIFOR = CPB.IDCLIFOR
                   AND CONTAS_PAGAR_BAIXAS.IDTITULO = CPB.IDTITULO
                   AND CONTAS_PAGAR_BAIXAS.DIGITOTITULO = CPB.DIGITOTITULO
                   AND CONTAS_PAGAR_BAIXAS.SERIENOTA = CPB.SERIENOTA
             )
     ) AS TMP 
    JOIN DBA.CONTABIL_PLANO_CONTAS AS CBEB ON CBEB.IDCTACONTABIL = TMP.IDCTADEBITO
